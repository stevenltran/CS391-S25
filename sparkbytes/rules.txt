rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // Allow only BU users
    function isBUUser() {
      return request.auth != null && request.auth.token.email.matches('.*@bu\\.edu$');
    }

    // Users can only access their own profile
    match /users/{userId} {
      allow read, write: if isBUUser() && request.auth.uid == userId;
    }

    // Events: Anyone at BU can read, only creator can modify
    match /events/{eventId} {
      allow read: if isBUUser();
      allow create: if isBUUser();
      allow update, delete: if isBUUser() && request.auth.uid == resource.data.creator;
    }

    // Notifications: Only the user can read/delete their own
    match /notifications/{notifId} {
      allow read, delete: if isBUUser() && request.auth.uid == resource.data.userId;
    }

    // Email sending (Trigger Email extension)
    match /mail/{mailId} {
      allow create: if isBUUser();
    }
  }
}
